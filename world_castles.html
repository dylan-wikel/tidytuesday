France has the most castles at 351. I subset the data to countries that contain 200 or more castles after viewing all of the data in with a simple bar plot.

Palaces vs. Fortresses

pvf_summary <- world_castles |>
  filter(category %in% c("palace", "fortress")) |>
  group_by(category) |>
  summarise(
    mean_year = round(mean(year, na.rm = TRUE)),
    median_year = median(year, na.rm = TRUE),
    quantile(year, 0.25, na.rm = TRUE),
    quantile(year, 0.75, na.rm = TRUE)
    )

pvf_boxplot <- world_castles |>
  filter(category %in% c("palace", "fortress")) |>
ggplot(aes(x = category, y = year)) +
  geom_boxplot()
